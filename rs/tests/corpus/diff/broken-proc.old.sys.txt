	func A(n: int): int;
	Good -> A(n = this.n);
}

	func A(n: int): int;
	Breaks -> A(n = this.n);
}

	func A(n: int): int;
	BadDecl -> A(n = this.n);
}

	func A(n: int): int;
	Always -> A(n = "no");
