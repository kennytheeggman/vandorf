
            type Ctx = { greeting: string };
            proc Setup(): Ctx { func S(): Ctx; Setup -> S(); }
            proc Hi(name: string, ctx: Ctx): string { func B(name: string, ctx: Ctx): string; Hi -> B(name = this.name, ctx = this.ctx); }
            type Req = { method: string, path: string, query: map[string]string, headers: map[string]string, body: string };
            proc Page(ctx: Ctx, req: Req): { status: int, headers: map[string]string, body: string } {
                func C(ctx: Ctx, req: Req): { status: int, headers: map[string]string, body: string };
                Page -> C(ctx = this.ctx, req = this.req);
            }
            proc NoCtx(name: string): string { func D(name: string): string; NoCtx -> D(name = this.name); }
            proc WrongCtx(ctx: string): string { func E(ctx: string): string; WrongCtx -> E(ctx = this.ctx); }
            proc Takes(n: int): Ctx { func F(n: int): Ctx; Takes -> F(n = this.n); }
